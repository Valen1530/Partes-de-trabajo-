<script>

// === CONFIGURA AQUÍ LA URL DE TU GOOGLE APPS SCRIPT (termina en /exec) ===

const API_URL =https://script.google.com/macros/s/AKfycbxYA53WV5wwEECvXpC8Q2BE4iWI87PTuKo6KA3PacOkcJniS2cSrWNVmcd0KSdJgIPx/exec;

// ==========================================================================

 

let catalogo = { precioHora: 0, materiales: [] };

 

function el(tag, cls, text){ const e = document.createElement(tag); if(cls) e.className = cls; if(text !== undefined) e.textContent = text; return e; }

 

function apiConfigured(){ return API_URL && !API_URL.startsWith('PEGA_AQUI'); }

 

async function apiGet(action){

  const res = await fetch(`${API_URL}?action=${action}`);

  return res.json();

}

async function apiPost(action, payload){

  const res = await fetch(API_URL, {

    method:'POST',

    headers:{'Content-Type':'text/plain;charset=utf-8'},

    body: JSON.stringify({action, payload})

  });

  return res.json();

}

 

document.querySelectorAll('.tab-btn').forEach(btn=>{

  btn.onclick = ()=>{

    document.querySelectorAll('.tab-btn').forEach(b=>b.classList.remove('active'));

    document.querySelectorAll('main section').forEach(s=>s.classList.remove('active'));

    btn.classList.add('active');

    document.getElementById(btn.dataset.tab).classList.add('active');

    if(btn.dataset.tab === 'historial') loadHistorial();

  };

});

 

async function loadCatalogo(){

  if(!apiConfigured()){ document.getElementById('apiwarn').style.display='block'; return; }

  try{

    catalogo = await apiGet('getCatalogo');

    if(!catalogo.materiales) catalogo.materiales = [];

  }catch(e){ console.error(e); }

  renderCatalog();

  renderMaterialSelects();

}

 

function renderCatalog(){

  document.getElementById('precioHora').value = catalogo.precioHora || '';

  const box = document.getElementById('catalogList');

  box.innerHTML = '';

  catalogo.materiales.forEach((m, i)=>{

    const row = el('div','material-row');

    const name = document.createElement('input');

    name.type='text'; name.value=m.name; name.placeholder='Material';

    name.oninput = ()=> catalogo.materiales[i].name = name.value;

    const price = document.createElement('input');

    price.type='number'; price.step='0.01'; price.value=m.price; price.placeholder='€/ud'; price.className='field-price';

    price.oninput = ()=> catalogo.materiales[i].price = parseFloat(price.value)||0;

    const del = el('button','icon-btn','×');

    del.onclick = ()=>{ catalogo.materiales.splice(i,1); renderCatalog(); };

    row.append(name, price, del);

    box.appendChild(row);

  });

}

 

function addCatalogRow(){ catalogo.materiales.push({name:'', price:0}); renderCatalog(); }

 

async function guardarCatalogo(){

  if(!apiConfigured()){ alert('Falta configurar API_URL en el código.'); return; }

  catalogo.precioHora = parseFloat(document.getElementById('precioHora').value) || 0;

  try{

    await apiPost('saveCatalogo', catalogo);

    renderMaterialSelects();

    alert('Precios guardados');

  }catch(e){ alert('No se pudo guardar. Revisa la conexión.'); }

}

 

function renderMaterialSelects(){

  document.querySelectorAll('.material-row select').forEach(sel=>{

    const current = sel.value;

    sel.innerHTML = '<option value="">Elige material…</option>' + catalogo.materiales.map(m=>`<option value="${m.name}">${m.name}</option>`).join('');

    sel.value = current;

  });

}

 

function addMaterialRow(listId){

  const row = el('div','material-row');

  const sel = document.createElement('select');

  sel.innerHTML = '<option value="">Elige material…</option>' + catalogo.materiales.map(m=>`<option value="${m.name}">${m.name}</option>`).join('');

  const qty = document.createElement('input');

  qty.type='number'; qty.min='0'; qty.step='1'; qty.placeholder='Cant.';

  const del = el('button','icon-btn','×');

  del.onclick = ()=> row.remove();

  row.append(sel, qty, del);

  document.getElementById(listId).appendChild(row);

}

 

function readMaterialRows(listId){

  const rows = [];

  document.querySelectorAll('#'+listId+' .material-row').forEach(row=>{

    const sel = row.querySelector('select');

    const qty = row.querySelector('input');

    const name = sel.value;

    const cantidad = parseFloat(qty.value) || 0;

    if(name && cantidad > 0){

      const priceInfo = catalogo.materiales.find(m=>m.name===name);

      rows.push({ name, cantidad, price: priceInfo ? priceInfo.price : 0 });

    }

  });

  return rows;

}

 

async function guardarParte(){

  if(!apiConfigured()){ alert('Falta configurar API_URL en el código.'); return; }

  const cliente = document.getElementById('cliente').value.trim();

  const fecha = document.getElementById('fecha').value || new Date().toISOString().slice(0,10);

  const trabajo = document.getElementById('trabajo').value.trim();

  const horas = parseFloat(document.getElementById('horas').value) || 0;

  const llevado = readMaterialRows('llevadoList');

  const devuelto = readMaterialRows('devueltoList');

  if(!cliente){ alert('Falta el nombre del cliente'); return; }

 

  const costeLlevado = llevado.reduce((s,m)=> s + m.cantidad*m.price, 0);

  const costeDevuelto = devuelto.reduce((s,m)=> s + m.cantidad*m.price, 0);

  const costeMaterial = costeLlevado - costeDevuelto;

  const costeHoras = horas * (catalogo.precioHora || 0);

  const total = costeMaterial + costeHoras;

  const id = Date.now().toString();

  const parte = { id, cliente, fecha, trabajo, horas, llevado, devuelto, costeMaterial, costeHoras, total };

 

  const btn = document.getElementById('guardarBtn');

  btn.disabled = true; btn.textContent = 'Guardando…';

  try{

    await apiPost('saveParte', parte);

    mostrarRecibo(parte);

    document.getElementById('cliente').value='';

    document.getElementById('trabajo').value='';

    document.getElementById('horas').value='';

    document.getElementById('llevadoList').innerHTML='';

    document.getElementById('devueltoList').innerHTML='';

  }catch(e){ alert('No se pudo guardar el parte. Revisa la conexión.'); }

  btn.disabled = false; btn.textContent = 'Guardar parte y generar factura';

}

 

function mostrarRecibo(p){

  const box = document.getElementById('reciboBox');

  box.style.display='block';

  const fmt = n => n.toLocaleString('es-ES', {minimumFractionDigits:2, maximumFractionDigits:2});

  let matHtml = '';

  p.llevado.forEach(m=>{ matHtml += `<div class="line"><span>${m.name} × ${m.cantidad}</span><span>${fmt(m.cantidad*m.price)} €</span></div>`; });

  if(p.devuelto.length){

    matHtml += `<div class="section-title">Devuelto (descontado)</div>`;

    p.devuelto.forEach(m=>{ matHtml += `<div class="line"><span>${m.name} × ${m.cantidad}</span><span>-${fmt(m.cantidad*m.price)} €</span></div>`; });

  }

  box.innerHTML = `

    <div class="receipt">

      <h2>PARTE / FACTURA</h2>

      <div class="meta">${p.fecha} · ${p.cliente}</div>

      <div class="section-title">Material</div>

      ${matHtml || '<div class="line"><span>—</span><span></span></div>'}

      <hr>

      <div class="line"><span>Horas (${p.horas} h)</span><span>${fmt(p.costeHoras)} €</span></div>

      <hr>

      <div class="line total"><span>TOTAL</span><span>${fmt(p.total)} €</span></div>

      ${p.trabajo ? `<div class="section-title">Trabajo realizado</div><div>${p.trabajo}</div>` : ''}

    </div>

    <button class="ghost-btn" onclick="window.print()">Imprimir / guardar como PDF</button>

  `;

}

 

async function loadHistorial(){

  const box = document.getElementById('histList');

  if(!apiConfigured()){ box.innerHTML = '<div class="empty">Configura API_URL para ver el historial</div>'; return; }

  box.innerHTML = '<div class="empty">Cargando…</div>';

  try{

    const partes = await apiGet('getPartes');

    if(!partes.length){ box.innerHTML = '<div class="empty">Todavía no hay partes guardados</div>'; return; }

    partes.sort((a,b)=> b.id.localeCompare(a.id));

    box.innerHTML = '';

    const fmt = n => n.toLocaleString('es-ES', {minimumFractionDigits:2, maximumFractionDigits:2});

    partes.forEach(p=>{

      const row = el('div','hist-item');

      row.innerHTML = `<div><div class="cli">${p.cliente}</div><div class="fecha">${p.fecha}</div></div><div class="total">${fmt(p.total)} €</div>`;

      row.onclick = ()=>{ document.querySelector('.tab-btn[data-tab="nuevo"]').click(); mostrarRecibo(p); };

      box.appendChild(row);

    });

  }catch(e){ box.innerHTML = '<div class="empty">No se pudo cargar el historial</div>'; }

}

 

document.getElementById('fecha').valueAsDate = new Date();

loadCatalogo();
